LCS(x, y):
    lcs <- inicializar(tamanho(x), tamanho(y), 0);

    para i ate tamanho(x):
        para j ate tamanho(y):
            // Caso base: Se ambos os arrays tiverem sido preenchidos, o tamanho de LCS sera 0
            se i = 0 ou j = 0:
                lcs[i][j] <- 0;

            // Se ambos arrays baterem
            senao se x[i - 1] = y[j - 1]:
                lcs[i][j] <- lcs[i - 1][j - 1] + 1;

            // Se os elementos de ambos os arrays nao baterem
            vizinho_cima <- lcs[i - 1][j];
            vizinho_baixo <- lcs[i][j - 1];

    // O valor contido na ultima celula da matriz e o tamanho da maior subsequencia
    tamanho_maior_subsequencia <- lcs[tamanho(x)][tamanho(y)];

    // Obtendo maior subsequencia
    // Criando um array para armazenar a maior subsequencia
    maior_subsequencia <- criar_array();

    // Realizando Backtracking da matriz utilizada na programacao dinamica
    i <- tamanho(x);
    j <- tamanho(y);
    enquanto i > 0 e j > 0:
        // Se o elemento da primeira serie for igual ao elemento da segunda serie na posicao atual
        se x[i - 1] = y[j - 1]:
            maior_subsequencia.inserir(0, [i - 1, j - 1]);
            i <- i - 1;
            j <- j - 1;

        // Se o elemento da primeira serie for diferente do elemento da segunda serie na posicao atual, sera analisado o maior valor
        senao se lcs[i - 1][j] > lcs[i][j - 1]:
            i <- i - 1;
            j <- j - 1;
        
    retorne tamanho_maior_subsequencia, maior_subsequencia;
