DTW(s, t, janela_busca):

    // Ajustando janela de busca
    se janela_busca for nula:
        janela_busca <- max(tamanho(s), tamanho(t))
    senão:
        janela_busca <- max(janela_busca, |tamanho(s) - tamanho(t)|)

    // Inicializando matriz dtw
    dtw <- criar_matriz(tamanho(s), tamanho(t))
    preencher_matriz(dtw, infinito)

    // Definindo condição inicial
    matrix_dtw[0, 0] <- 0

    // Preenchendo a matriz por Programação dinâmica
    para i de 1 até tamanho(s) + 1:
        // Definido os limites inferior e superior
        j_inicio <- max(1, i - janela_busca)
        j_fim <- min(tamanho(t), i + janela_busca) + 1

        para j de j_inicio até j_fim:
            // Calculando o custo local
            custo_local <- distancia(s[i - 1], t[j - 1])

            // Obtendo o menor vizinho
            vizinho_superior <- dtw[i - 1, j]
            vizinho_esquerdo <- dtw[i, j - 1]
            vizinho_diagonal <- dtw[i - 1, j - 1]
            menor_vizinho <- min(vizinho_superior, vizinho_esquerdo, vizinho_diagonal)

            // Definindo o valor da posição atual
            dtw[i, j] <- custo_local + menor_vizinho

        // Distância final
        distancia = dtw[tamanho(s), tamanho(t)]

    retorne distancia, dtw



int DTWDistance(s: array [1..n], t: array [1..m], w: int) {
    DTW := array [0..n, 0..m]

    w := max(w, abs(n-m)) // adapt window size (*)

    for i := 0 to n
        for j:= 0 to m
            DTW[i, j] := infinity
    DTW[0, 0] := 0
    for i := 1 to n
        for j := max(1, i-w) to min(m, i+w)
            DTW[i, j] := 0

    for i := 1 to n
        for j := max(1, i-w) to min(m, i+w)
            cost := d(s[i], t[j])
            DTW[i, j] := cost + minimum(DTW[i-1, j  ],    // insertion
                                        DTW[i  , j-1],    // deletion
                                        DTW[i-1, j-1])    // match
    return DTW[n, m]
}
